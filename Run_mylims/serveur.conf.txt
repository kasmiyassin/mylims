<VirtualHost *:80>
    ServerName 134.110.12.162
    ServerAdmin webmaster@localhost
    DocumentRoot /var/www/html

    ErrorLog  ${APACHE_LOG_DIR}/mylims-error.log
    CustomLog ${APACHE_LOG_DIR}/mylims-access.log combined

    ProxyRequests Off
    ProxyPreserveHost On
 
    # --- LIMS  Proxy (Port 5300 - Working) ---
    <Location "/mylims/">
        ProxyPass         http://127.0.0.1:5300/
        ProxyPassReverse  http://127.0.0.1:5300/
        RequestHeader set X-Forwarded-Proto "http"
        Header edit Set-Cookie "(.*Path=/)(;.*)" "$1mylims/$2"
    </Location>
    ProxyPassReverseCookiePath / /mylims/


    # --- LIMS demo  (Port 5400) ---
    <Location "/mylims_demo/">
        ProxyPass         http://127.0.0.1:5400/
        ProxyPassReverse  http://127.0.0.1:5400/
        RequestHeader set X-Forwarded-Proto "http"
        Header edit Set-Cookie "(.*Path=/)(;.*)" "$1mylims_demo/$2"
    </Location>
    ProxyPassReverseCookiePath / /mylims_demo/

</VirtualHost>
